  const API_URL = "https://script.google.com/macros/s/AKfycbyBvgx4a1rtfuCfx0-h5WEP1Pb82DvVMFcPhBOAQY9dJiJ197daXpgkfWJUBS02b5Q_/exec";

  async function callBackend(action, payload = {}) {
    return new Promise((resolve, reject) => {
      // Cek apakah web berjalan di dalam iframe GAS
      if (typeof google !== 'undefined' && google.script && google.script.run) {
        let runner = google.script.run
          .withSuccessHandler(res => resolve(res))
          .withFailureHandler(err => reject(err));
          
        if (action === "getInitialData") runner.getInitialData();
        else if (action === "validasiPinKeServer") runner.validasiPinKeServer(payload.inputPin);
        else if (action === "cariData") runner.cariData(payload.keyword, payload.startDate, payload.endDate);
        else if (action === "getLeaderboard") runner.getLeaderboard(payload.startDate, payload.endDate);
        else if (action === "ajukanPengawas") runner.ajukanPengawas(payload.data);
        else if (action === "updatePrerogatif") runner.updatePrerogatif(payload.keyword, payload.isGranted);
        else if (action === "saveTempImage") runner.saveTempImage(payload.base64Data, payload.fileName);
      } else {
        // Jika berjalan di GitHub / Jotform / Hosting Eksternal, gunakan fetch API
        fetch(API_URL, { 
          method: "POST", 
          // PROTEKSI CORS: Memaksa pengiriman dengan tipe text/plain agar Google tidak menolak preflight OPTIONS
          headers: { "Content-Type": "text/plain;charset=utf-8" },
          body: JSON.stringify({ action, ...payload }) 
        })
        .then(res => res.json())
        .then(json => json.status === "success" ? resolve(json.data) : reject(new Error(json.message)))
        .catch(reject);
      }
    });
  }

        if(hse) Object.assign(hse.style, { animation:'', opacity:'' });

        if (bList) bList.style.cssText = oriBListStyle; 
        $('raportHeader').style.display='none'; vFooter.style.display='none'; btn.style.display='flex';
        
        let imgData = canvas.toDataURL("image/jpeg", 1.0);
        let fName = `Raport_${($('uName').innerText||"SAP").replace(/\W/g,"_")}.jpg`;
        
        const isMobileDevice = /Android|webOS|iPhone|iPad|iPod|BlackBerry|IEMobile|Opera Mini/i.test(navigator.userAgent);
        const isNativeApp = !!window.ReactNativeWebView;

        if (isMobileDevice || isNativeApp) {
          btn.innerText = '⏳ Menyimpan...';
          callBackend("saveTempImage", { base64Data: imgData.split(',')[1], fileName: fName }).then(url => {
            if(window.ThunkableWebviewerExtension) {
               window.ThunkableWebviewerExtension.postMessage(url);
            } else if (window.ReactNativeWebView) {
               window.ReactNativeWebView.postMessage(url);
            } else {
               let link = document.createElement('a');
               link.href = url;
               link.target = '_blank';
               document.body.appendChild(link);
               link.click();
               document.body.removeChild(link);
            }
            btn.innerText='✅ Berhasil!'; setTimeout(()=> btn.innerText='📥 Download Raport', 3000); 
          }).catch(e => { alert("Gagal upload."); btn.innerText='📥 Download Raport'; });
        } else {
          fetch(imgData).then(res => res.blob()).then(blob => {
            let blobUrl = URL.createObjectURL(blob);
            let a = document.createElement('a');
            a.href = blobUrl; a.download = fName;
            document.body.appendChild(a); a.click(); document.body.removeChild(a);
            URL.revokeObjectURL(blobUrl);
          });
        }
      }).catch(e => { alert("Gagal merender gambar."); btn.style.display='flex'; Object.assign(resDiv.style, oriStyle); });
    }, 150);
  };
